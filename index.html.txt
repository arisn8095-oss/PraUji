        class GoogleSheetService {
            constructor() {
                this.sheetId = "1jc5nOgZOQg0BV-KzQikKFcJ3-s2CiukfJGtjWg221FM";
                this.tabSiswa = "SISWA";

            async init() {

            async fetchSiswaData() {
                const statusEl = document.getElementById('login-sheet-status');
                if (statusEl) {
                    statusEl.innerHTML = `<i class="fa-solid fa-spinner fa-spin text-amber-400"></i> <span>Menghubungkan ke Google Sheet (SISWA)...</span>`;
                }

                // Jika Web App URL Google Apps Script diatur
                if (this.scriptUrl) {
                    try {
                        const res = await fetch(`${this.scriptUrl}?action=getSiswa`);
                        const json = await res.json();
                        if (json.status === "success" && json.data) {
                            this.siswaList = json.data;
                            state.roster = this.siswaList.map(s => s.nama);
                            state.save();
                            if (statusEl) {
                                statusEl.innerHTML = `<i class="fa-solid fa-circle-check text-emerald-400"></i> <span class="text-emerald-300">${this.siswaList.length} Data Siswa Terhubung via Apps Script!</span>`;
                            }
                            return;
                        }
                    } catch (err) {
                        console.warn("Error fetching via Web App URL, falling back to Google Visualization API...", err);
                    }
                }

                // Fallback direct read via GViz API
                try {
                    const gvizUrl = `https://docs.google.com/spreadsheets/d/${this.sheetId}/gviz/tq?tqx=out:json&sheet=${encodeURIComponent(this.tabSiswa)}`;
                    const res = await fetch(gvizUrl);
                    const text = await res.text();
                    
                    const jsonMatch = text.match(/google\.visualization\.Query\.setResponse\(([\s\S]*)\);/);
                    if (jsonMatch && jsonMatch[1]) {
                        const parsed = JSON.parse(jsonMatch[1]);
                        const rows = parsed.table.rows || [];
                        
                        this.siswaList = rows.map((r, idx) => {
                            const c = r.c || [];
                            return {
                                nis: c[0] ? String(c[0].v || c[0].f || "").trim() : "",
                                nama: c[1] ? String(c[1].v || "").trim() : `Siswa #${idx+1}`,
                                kelas: c[2] ? String(c[2].v || "").trim() : "TIK",
                                noAbsen: c[3] ? parseInt(c[3].v) : (idx + 1)
                            };
                        }).filter(s => s.nis.length > 0 || s.nama.length > 0);

                        if (this.siswaList.length > 0) {
                            state.roster = this.siswaList.map(s => s.nama);
                            state.save();
                            if (statusEl) {
                                statusEl.innerHTML = `<i class="fa-solid fa-circle-check text-emerald-400"></i> <span class="text-emerald-300">${this.siswaList.length} Data Siswa Terhubung dari Sheet!</span>`;
                            }
                            return;
                        }
                    }
                } catch (e) {
                    console.warn("Gviz API read error, using offline/local cache", e);
                }

                if (statusEl) {
                    statusEl.innerHTML = `<i class="fa-solid fa-wifi text-amber-400"></i> <span>Mode Standalone / Siap Login NIS</span>`;
                }
            }

        class SoundFX {
            constructor() {
                this.enabled = true;
                this.ctx = null;
            }

            init() {
                if (!this.ctx) {
                    const AudioCtx = window.AudioContext || window.webkitAudioContext;
                    if (AudioCtx) this.ctx = new AudioCtx();
                }
            }

            toggleSound() {
                this.enabled = !this.enabled;
                const btn = document.getElementById('sound-btn');
                if (this.enabled) {
                    btn.innerHTML = '<i class="fa-solid fa-volume-high text-sm"></i>';
                    btn.classList.remove('opacity-50');
                    this.playTone(600, 'sine', 0.1, 0.1);
                } else {
                    btn.innerHTML = '<i class="fa-solid fa-volume-xmark text-sm"></i>';
                    btn.classList.add('opacity-50');
                }
            }

            playTone(freq, type, duration, vol = 0.1) {
                if (!this.enabled) return;
                this.init();
                if (!this.ctx) return;

                try {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = type;
                    osc.frequency.value = freq;
                    gain.gain.setValueAtTime(vol, this.ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.0001, this.ctx.currentTime + duration);
                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start();
                    osc.stop(this.ctx.currentTime + duration);
                } catch (e) {}
            }

            playSnap() {
                if (!this.enabled) return;
                this.init();
                try {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(350, this.ctx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(80, this.ctx.currentTime + 0.08);
                    gain.gain.setValueAtTime(0.25, this.ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + 0.08);
                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start();
                    osc.stop(this.ctx.currentTime + 0.08);
                } catch(e) {}
            }

            playWoodHit() {
                if (!this.enabled) return;
                this.init();
                try {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(160, this.ctx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(40, this.ctx.currentTime + 0.15);
                    gain.gain.setValueAtTime(0.4, this.ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + 0.15);
                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start();
                    osc.stop(this.ctx.currentTime + 0.15);
                    this.playTone(800, 'square', 0.02, 0.2);
                } catch(e) {}
            }

            playBounce() {
                if (!this.enabled) return;
                this.playTone(280, 'sine', 0.06, 0.1);
            }

            playCheer() {
                if (!this.enabled) return;
                const freqs = [523.25, 659.25, 783.99, 1046.50];
                freqs.forEach((f, i) => {
                    setTimeout(() => this.playTone(f, 'triangle', 0.2, 0.15), i * 70);
                });
            }

            playCorrect() {
                if (!this.enabled) return;
                this.playTone(523.25, 'sine', 0.12, 0.15);
                setTimeout(() => this.playTone(659.25, 'sine', 0.12, 0.15), 100);
                setTimeout(() => this.playTone(783.99, 'sine', 0.25, 0.2), 200);
            }

            playWrong() {
                if (!this.enabled) return;
                this.playTone(220, 'sawtooth', 0.15, 0.15);
                setTimeout(() => this.playTone(180, 'sawtooth', 0.25, 0.2), 120);
            }

            playClick() {
                this.playTone(400, 'sine', 0.05, 0.05);
            }

            playLevelUp() {
                if (!this.enabled) return;
                const notes = [440, 554.37, 659.25, 880];
                notes.forEach((freq, idx) => {
                    setTimeout(() => this.playTone(freq, 'triangle', 0.2, 0.15), idx * 100);
                });
            }
        }

        /* ==========================================================================
           5. MINIGAME: LEMPAR NOMOR ABSEN (REALISTIC 2D GRAVITY & SLINGSHOT ENGINE)
           ========================================================================== */
        class ThrowAbsenEngine {
            constructor() {
                this.studentCount = 20;
                this.usedAbsen = new Set();
                this.canvas = null;
                this.ctx = null;
                this.targets = [];
                this.particles = [];
                this.projectile = null;
                this.animFrame = null;
                
                this.isDragging = false;
                this.dragStart = { x: 0, y: 0 };
                this.dragCurrent = { x: 0, y: 0 };
                this.gravity = 0.38;
            }

            startMode() {
                app.showScreen('screen-throw-absen');
                setTimeout(() => {
                    this.initCanvas();
                    this.buildTargets();
                    this.renderStatusGrid();
                }, 50);
            }

            setStudentCount(count) {
                this.studentCount = parseInt(count);
                this.resetUsedAbsen();
            }

            resetUsedAbsen() {
                this.usedAbsen.clear();
                this.buildTargets();
                this.renderStatusGrid();
                app.showInfoMsg('Daftar nomor absen terpanggil telah di-reset!');
            }

            initCanvas() {
                this.canvas = document.getElementById('target-canvas');
                this.ctx = this.canvas.getContext('2d');

                const rect = this.canvas.getBoundingClientRect();
                this.canvas.width = rect.width;
                this.canvas.height = rect.height;

                this.canvas.onmousedown = (e) => this.handlePointerDown(e);
                this.canvas.onmousemove = (e) => this.handlePointerMove(e);
                this.canvas.onmouseup = () => this.handlePointerUp();
                this.canvas.onmouseleave = () => this.handlePointerUp();

                this.canvas.ontouchstart = (e) => this.handlePointerDown(e.touches[0]);
                this.canvas.ontouchmove = (e) => this.handlePointerMove(e.touches[0]);
                this.canvas.ontouchend = () => this.handlePointerUp();

                if (!this.animFrame) {
                    this.loop();
                }
            }

            buildTargets() {
                if (!this.canvas) return;
                this.targets = [];
                const cols = 5;
                const rows = Math.ceil(this.studentCount / cols);
                const paddingX = this.canvas.width / (cols + 1);
                const paddingY = (this.canvas.height * 0.5) / (rows + 1);

                for (let i = 1; i <= this.studentCount; i++) {
                    const row = Math.floor((i - 1) / cols);
                    const col = (i - 1) % cols;
                    const baseX = paddingX * (col + 1);
                    const baseY = 45 + paddingY * (row + 1);

                    this.targets.push({
                        num: i,
                        x: baseX,
                        y: baseY,
                        radius: Math.min(24, paddingX * 0.38),
                        vx: (Math.random() - 0.5) * 0.5,
                        vy: (Math.random() - 0.5) * 0.3,
                        rotation: 0,
                        rotSpeed: 0,
                        isHit: this.usedAbsen.has(i),
                        falling: false
                    });
                }
            }

            getSlingshotPos() {
                return {
                    x: this.canvas.width / 2,
                    y: this.canvas.height - 55
                };
            }

            handlePointerDown(e) {
                if (this.projectile) return;
                const rect = this.canvas.getBoundingClientRect();
                const clickX = e.clientX - rect.left;
                const clickY = e.clientY - rect.top;

                const base = this.getSlingshotPos();
                const dist = Math.hypot(clickX - base.x, clickY - base.y);

                if (dist < 70) {
                    this.isDragging = true;
                    this.dragStart = { x: base.x, y: base.y };
                    this.dragCurrent = { x: clickX, y: clickY };
                }
            }

            handlePointerMove(e) {
                if (!this.isDragging) return;
                const rect = this.canvas.getBoundingClientRect();
                const clickX = e.clientX - rect.left;
                const clickY = e.clientY - rect.top;

                const base = this.getSlingshotPos();
                const maxDrag = 90;
                const dx = clickX - base.x;
                const dy = clickY - base.y;
                const len = Math.hypot(dx, dy);

                if (len > maxDrag) {
                    this.dragCurrent = {
                        x: base.x + (dx / len) * maxDrag,
                        y: base.y + (dy / len) * maxDrag
                    };
                } else {
                    this.dragCurrent = { x: clickX, y: clickY };
                }
            }

            handlePointerUp() {
                if (!this.isDragging) return;
                this.isDragging = false;

                const base = this.getSlingshotPos();
                const dx = base.x - this.dragCurrent.x;
                const dy = base.y - this.dragCurrent.y;

                if (Math.hypot(dx, dy) < 15) return;

                sound.playSnap();

                const powerScale = 0.22;
                this.projectile = {
                    x: base.x,
                    y: base.y,
                    vx: dx * powerScale,
                    vy: dy * powerScale,
                    radius: 12,
                    rotation: 0
                };

                document.getElementById('throw-status-msg').innerText = 'Bola meluncur!';
            }

            update() {
                if (!this.canvas) return;

                this.targets.forEach(t => {
                    if (t.falling) {
                        t.y += t.vy;
                        t.vy += this.gravity;
                        t.x += t.vx;
                        t.rotation += t.rotSpeed;
                    } else {
                        t.x += t.vx;
                        t.y += t.vy;
                        if (t.x - t.radius < 10 || t.x + t.radius > this.canvas.width - 10) t.vx *= -1;
                        if (t.y - t.radius < 20 || t.y + t.radius > this.canvas.height * 0.6) t.vy *= -1;
                    }
                });

                if (this.projectile) {
                    const p = this.projectile;
                    p.x += p.vx;
                    p.y += p.vy;
                    p.vy += this.gravity;
                    p.rotation += p.vx * 0.05;

                    if (p.x - p.radius < 0) {
                        p.x = p.radius;
                        p.vx *= -0.75;
                        sound.playBounce();
                    } else if (p.x + p.radius > this.canvas.width) {
                        p.x = this.canvas.width - p.radius;
                        p.vx *= -0.75;
                        sound.playBounce();
                    }

                    if (p.y - p.radius < 0) {
                        p.y = p.radius;
                        p.vy *= -0.75;
                        sound.playBounce();
                    }

                    for (let t of this.targets) {
                        if (t.isHit || t.falling) continue;

                        const dist = Math.hypot(p.x - t.x, p.y - t.y);
                        if (dist < p.radius + t.radius) {
                            sound.playWoodHit();
                            sound.playCheer();

                            t.falling = true;
                            t.vy = -3;
                            t.vx = p.vx * 0.3;
                            t.rotSpeed = (Math.random() - 0.5) * 0.2;

                            this.spawnSplinters(t.x, t.y);
                            this.triggerAbsenHit(t.num);
                            this.projectile = null;
                            return;
                        }
                    }

                    if (p.y > this.canvas.height + 30) {
                        this.projectile = null;
                        document.getElementById('throw-status-msg').innerText = 'Lemparan meleset! Tarik & tembak lagi.';
                    }
                }

                for (let i = this.particles.length - 1; i >= 0; i--) {
                    const pt = this.particles[i];
                    pt.x += pt.vx;
                    pt.y += pt.vy;
                    pt.vy += 0.2;
                    pt.alpha -= 0.02;
                    pt.rot += pt.rotSpeed;

                    if (pt.alpha <= 0) {
                        this.particles.splice(i, 1);
                    }
                }
            }

            spawnSplinters(x, y) {
                for (let i = 0; i < 16; i++) {
                    const angle = Math.random() * Math.PI * 2;
                    const speed = Math.random() * 6 + 2;
                    this.particles.push({
                        x: x,
                        y: y,
                        vx: Math.cos(angle) * speed,
                        vy: Math.sin(angle) * speed - 2,
                        size: Math.random() * 6 + 3,
                        color: Math.random() > 0.5 ? '#b45309' : '#f59e0b',
                        alpha: 1,
                        rot: Math.random() * Math.PI,
                        rotSpeed: (Math.random() - 0.5) * 0.3
                    });
                }
            }

            }

            draw() {
                if (!this.canvas || !this.ctx) return;
                const ctx = this.ctx;
                ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

                const base = this.getSlingshotPos();

                ctx.fillStyle = '#0f172a';
                ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);

                ctx.strokeStyle = 'rgba(255, 255, 255, 0.05)';
                ctx.lineWidth = 2;
                for (let y = 80; y < this.canvas.height * 0.65; y += 70) {
                    ctx.beginPath();
                    ctx.moveTo(10, y);
                    ctx.lineTo(this.canvas.width - 10, y);
                    ctx.stroke();
                }

                this.targets.forEach(t => {
                    ctx.save();
                    ctx.translate(t.x, t.y);
                    ctx.rotate(t.rotation);

                    ctx.beginPath();
                    ctx.arc(3, 4, t.radius, 0, Math.PI * 2);
                    ctx.fillStyle = 'rgba(0,0,0,0.4)';
                    ctx.fill();

                    ctx.beginPath();
                    ctx.arc(0, 0, t.radius, 0, Math.PI * 2);
                    if (t.isHit) {
                        ctx.fillStyle = '#334155';
                        ctx.strokeStyle = '#475569';
                    } else {
                        const grad = ctx.createRadialGradient(-3, -3, 2, 0, 0, t.radius);
                        grad.addColorStop(0, '#f59e0b');
                        grad.addColorStop(0.7, '#d97706');
                        grad.addColorStop(1, '#78350f');
                        ctx.fillStyle = grad;
                        ctx.strokeStyle = '#fef08a';
                    }
                    ctx.lineWidth = 3;
                    ctx.fill();
                    ctx.stroke();

                    ctx.beginPath();
                    ctx.arc(0, 0, t.radius * 0.65, 0, Math.PI * 2);
                    ctx.strokeStyle = t.isHit ? 'rgba(255,255,255,0.1)' : 'rgba(255,255,255,0.4)';
                    ctx.lineWidth = 1.5;
                    ctx.stroke();

                    ctx.fillStyle = t.isHit ? '#94a3b8' : '#ffffff';
                    ctx.font = `bold ${t.radius * 0.85}px 'Outfit', sans-serif`;
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillText(`#${t.num}`, 0, 0);

                    ctx.restore();
                });

                this.particles.forEach(pt => {
                    ctx.save();
                    ctx.translate(pt.x, pt.y);
                    ctx.rotate(pt.rot);
                    ctx.fillStyle = pt.color;
                    ctx.globalAlpha = pt.alpha;
                    ctx.fillRect(-pt.size / 2, -pt.size / 2, pt.size, pt.size);
                    ctx.restore();
                });

                if (this.isDragging) {
                    const dx = base.x - this.dragCurrent.x;
                    const dy = base.y - this.dragCurrent.y;
                    let simX = base.x;
                    let simY = base.y;
                    let simVx = dx * 0.22;
                    let simVy = dy * 0.22;

                    ctx.fillStyle = '#f59e0b';
                    for (let t = 0; t < 22; t++) {
                        simX += simVx;
                        simY += simVy;
                        simVy += this.gravity;

                        ctx.beginPath();
                        ctx.arc(simX, simY, 3, 0, Math.PI * 2);
                        ctx.globalAlpha = 1 - (t / 22);
                        ctx.fill();
                    }
                    ctx.globalAlpha = 1.0;
                }

                const forkLeft = { x: base.x - 22, y: base.y - 10 };
                const forkRight = { x: base.x + 22, y: base.y - 10 };
                const ballPos = this.isDragging ? this.dragCurrent : base;

                if (this.isDragging) {
                    ctx.beginPath();
                    ctx.moveTo(forkLeft.x, forkLeft.y);
                    ctx.lineTo(ballPos.x, ballPos.y);
                    ctx.strokeStyle = '#b45309';
                    ctx.lineWidth = 4;
                    ctx.stroke();
                }

                if (!this.projectile) {
                    ctx.beginPath();
                    ctx.arc(ballPos.x, ballPos.y + 4, 12, 0, Math.PI * 2);
                    ctx.fillStyle = 'rgba(0,0,0,0.3)';
                    ctx.fill();

                    const ballGrad = ctx.createRadialGradient(ballPos.x - 3, ballPos.y - 3, 2, ballPos.x, ballPos.y, 12);
                    ballGrad.addColorStop(0, '#fef08a');
                    ballGrad.addColorStop(0.5, '#f59e0b');
                    ballGrad.addColorStop(1, '#b45309');

                    ctx.beginPath();
                    ctx.arc(ballPos.x, ballPos.y, 12, 0, Math.PI * 2);
                    ctx.fillStyle = ballGrad;
                    ctx.fill();
                    ctx.strokeStyle = '#ffffff';
                    ctx.lineWidth = 1.5;
                    ctx.stroke();
                }

                if (this.isDragging) {
                    ctx.beginPath();
                    ctx.moveTo(forkRight.x, forkRight.y);
                    ctx.lineTo(ballPos.x, ballPos.y);
                    ctx.strokeStyle = '#d97706';
                    ctx.lineWidth = 4;
                    ctx.stroke();
                }

                ctx.fillStyle = '#78350f';
                ctx.strokeStyle = '#451a03';
                ctx.lineWidth = 3;

                ctx.beginPath();
                ctx.moveTo(base.x - 6, base.y + 35);
                ctx.lineTo(base.x - 6, base.y);
                ctx.lineTo(forkLeft.x, forkLeft.y);
                ctx.lineTo(forkLeft.x + 8, forkLeft.y);
                ctx.lineTo(base.x + 2, base.y + 10);
                ctx.lineTo(forkRight.x - 8, forkRight.y);
                ctx.lineTo(forkRight.x, forkRight.y);
                ctx.lineTo(base.x + 6, base.y);
                ctx.lineTo(base.x + 6, base.y + 35);
                ctx.closePath();
                ctx.fill();
                ctx.stroke();

                if (this.projectile) {
                    const p = this.projectile;
                    ctx.save();
                    ctx.translate(p.x, p.y);
                    ctx.rotate(p.rotation);

                    const pGrad = ctx.createRadialGradient(-3, -3, 2, 0, 0, p.radius);
                    pGrad.addColorStop(0, '#fef08a');
                    pGrad.addColorStop(0.6, '#f59e0b');
                    pGrad.addColorStop(1, '#b45309');

                    ctx.beginPath();
                    ctx.arc(0, 0, p.radius, 0, Math.PI * 2);
                    ctx.fillStyle = pGrad;
                    ctx.shadowColor = '#f59e0b';
                    ctx.shadowBlur = 12;
                    ctx.fill();
                    ctx.shadowBlur = 0;
                    ctx.strokeStyle = '#ffffff';
                    ctx.lineWidth = 1.5;
                    ctx.stroke();

                    ctx.restore();
                }
            }

            loop() {
                this.update();
                this.draw();
                this.animFrame = requestAnimationFrame(() => this.loop());
            }

        class GameEngine {
            constructor() {
                this.mode = 'adventure';
                this.questions = [];
                this.currentIdx = 0;
                this.score = 0;
                this.lives = 3;
                this.timer = 20;
                this.maxTimer = 20;
                this.timerInterval = null;
                this.userAnswers = [];
                this.selectedSubject = null;
                this.currentAbsenNum = null;
                this.powerups = {
                    p5050: true,
                    pTime: true,
                    pHint: true
                };
            }

            startAdventure() {
                this.mode = 'adventure';
                this.questions = this.shuffleArray([...QUESTIONS_DB]).slice(0, 10);
                this.initQuiz();
            }

            startSpeedRun() {
                this.mode = 'speedrun';
                this.questions = this.shuffleArray([...QUESTIONS_DB]);
                this.lives = 3;
                this.initQuiz();
            }

            startPractice(category) {
                this.mode = 'practice';
                this.selectedSubject = category;
                const filtered = QUESTIONS_DB.filter(q => q.category === category);
                this.questions = this.shuffleArray([...filtered]);
                this.initQuiz();
            }

            startAbsenQuiz(absenNum, singleQuestion) {
                this.mode = 'absen';
                this.currentAbsenNum = absenNum;
                this.questions = [singleQuestion];
                this.initQuiz();
            }

            restartCurrentMode() {
                if (this.mode === 'adventure') this.startAdventure();
                else if (this.mode === 'speedrun') this.startSpeedRun();
                else if (this.mode === 'practice') this.startPractice(this.selectedSubject);
                else if (this.mode === 'absen') throwGame.startMode();
            }

            initQuiz() {
                this.currentIdx = 0;
                this.score = 0;
                this.userAnswers = [];
                this.powerups = { p5050: true, pTime: true, pHint: true };

                app.showScreen('screen-quiz');
                this.renderPowerups();
                this.loadQuestion();
            }

            loadQuestion() {
                clearInterval(this.timerInterval);
                const q = this.questions[this.currentIdx];

                document.getElementById('feedback-card').classList.add('hidden');
                document.getElementById('hint-box').classList.add('hidden');
                
                document.getElementById('quiz-category-badge').innerText = q.category;
                document.getElementById('quiz-current-idx').innerText = this.currentIdx + 1;
                document.getElementById('quiz-total-idx').innerText = this.questions.length;
                document.getElementById('quiz-score-text').innerText = this.score;

                const absenTag = document.getElementById('quiz-absen-tag');
                if (this.mode === 'absen' && this.currentAbsenNum) {
                    absenTag.classList.remove('hidden');
                    document.getElementById('quiz-absen-num').innerText = this.currentAbsenNum;
                } else {
                    absenTag.classList.add('hidden');
                }

                const progPct = ((this.currentIdx + 1) / this.questions.length) * 100;
                document.getElementById('quiz-progress-bar').style.width = `${progPct}%`;

                const livesCont = document.getElementById('quiz-lives-container');
                if (this.mode === 'speedrun') {
                    livesCont.classList.remove('hidden');
                    document.getElementById('quiz-lives-text').innerText = this.lives;
                } else {
                    livesCont.classList.add('hidden');
                }

                document.getElementById('question-text').innerText = q.question;

                const container = document.getElementById('options-container');
                container.innerHTML = '';

                const optionLabels = ['A', 'B', 'C', 'D'];
                q.options.forEach((opt, idx) => {
                    const btn = document.createElement('button');
                    btn.className = `option-btn w-full p-4 rounded-xl glass-card text-left flex items-center justify-between border border-slate-700/80 hover:border-indigo-500 hover:bg-slate-800/90 transition-all text-slate-200 text-sm font-medium group`;
                    btn.dataset.idx = idx;
                    btn.onclick = () => this.handleAnswer(idx);

                    btn.innerHTML = `
                        <div class="flex items-center space-x-3">
                            <span class="w-7 h-7 rounded-lg bg-slate-800 border border-slate-600 group-hover:border-indigo-400 group-hover:bg-indigo-600/30 font-mono font-bold text-xs flex items-center justify-center text-slate-300 group-hover:text-indigo-300 transition-colors">
                                ${optionLabels[idx]}
                            </span>
                            <span>${opt}</span>
                        </div>
                        <i class="fa-regular fa-circle text-slate-600 group-hover:text-indigo-400"></i>
                    `;
                    container.appendChild(btn);
                });

                this.timer = 20;
                this.updateTimerDisplay();
                this.startTimer();
            }

            startTimer() {
                this.timerInterval = setInterval(() => {
                    this.timer--;
                    this.updateTimerDisplay();

                    if (this.timer <= 0) {
                        clearInterval(this.timerInterval);
                        sound.playWrong();
                        this.handleAnswer(-1);
                    }
                }, 1000);
            }

            updateTimerDisplay() {
                const el = document.getElementById('quiz-timer-text');
                el.innerText = `${this.timer}s`;
                if (this.timer <= 5) {
                    el.className = 'font-mono font-bold text-sm text-red-400 animate-pulse';
                } else {
                    el.className = 'font-mono font-bold text-sm text-amber-300';
                }
            }

            handleAnswer(selectedIdx) {
                clearInterval(this.timerInterval);
                const q = this.questions[this.currentIdx];
                const isCorrect = (selectedIdx === q.answer);
                const timeTaken = 20 - Math.max(0, this.timer);

                const optBtns = document.querySelectorAll('.option-btn');
                optBtns.forEach(btn => btn.disabled = true);

                if (isCorrect) {
                    sound.playCorrect();
                    const addedScore = 100 + (this.timer * 5);
                    this.score += addedScore;
                    document.getElementById('quiz-score-text').innerText = this.score;

                    if (selectedIdx >= 0) {
                        optBtns[selectedIdx].className = 'option-btn w-full p-4 rounded-xl bg-emerald-950/60 border-2 border-emerald-500 text-emerald-200 text-sm font-medium flex items-center justify-between';
                        optBtns[selectedIdx].querySelector('i').className = 'fa-solid fa-circle-check text-emerald-400';
                    }
                } else {
                    sound.playWrong();
                    if (selectedIdx >= 0) {
                        optBtns[selectedIdx].className = 'option-btn w-full p-4 rounded-xl bg-red-950/60 border-2 border-red-500 text-red-200 text-sm font-medium flex items-center justify-between';
                        optBtns[selectedIdx].querySelector('i').className = 'fa-solid fa-circle-xmark text-red-400';
                    }
                    optBtns[q.answer].className = 'option-btn w-full p-4 rounded-xl bg-emerald-950/40 border border-emerald-500/80 text-emerald-200 text-sm font-medium flex items-center justify-between';

                    if (this.mode === 'speedrun') {
                        this.lives--;
                        document.getElementById('quiz-lives-text').innerText = this.lives;
                    }
                }

                this.userAnswers.push({
                    question: q,
                    userSelected: selectedIdx,
                    isCorrect: isCorrect,
                    timeTaken: timeTaken
                });

                this.showFeedback(isCorrect, q);
            }

            showFeedback(isCorrect, q) {
                const card = document.getElementById('feedback-card');
                const title = document.getElementById('feedback-title');
                const exp = document.getElementById('feedback-explanation');

                card.classList.remove('hidden', 'border-emerald-500/40', 'border-red-500/40');

                if (isCorrect) {
                    card.classList.add('border-emerald-500/40');
                    title.className = 'font-heading font-bold text-lg text-emerald-400 flex items-center gap-2';
                    title.innerHTML = `<i class="fa-solid fa-circle-check"></i> Jawaban Tepat! (+100 Poin)`;
                } else {
                    card.classList.add('border-red-500/40');
                    title.className = 'font-heading font-bold text-lg text-red-400 flex items-center gap-2';
                    title.innerHTML = `<i class="fa-solid fa-triangle-exclamation"></i> Kurang Tepat! Kunci: (${['A','B','C','D'][q.answer]}) ${q.options[q.answer]}`;
                }

                exp.innerText = q.explanation;
            }

            nextQuestion() {
                sound.playClick();
                this.currentIdx++;

                if (this.mode === 'speedrun' && this.lives <= 0) {
                    this.finishQuiz();
                    return;
                }

                if (this.currentIdx >= this.questions.length) {
                    if (this.mode === 'absen') {
                        throwGame.startMode();
                    } else {
                        this.finishQuiz();
                    }
                } else {
                    this.loadQuestion();
                }
            }

            usePowerup5050() {
                if (!this.powerups.p5050) return;
                sound.playClick();
                this.powerups.p5050 = false;
                this.renderPowerups();

                const q = this.questions[this.currentIdx];
                const wrongIndices = [0, 1, 2, 3].filter(i => i !== q.answer);
                const shuffledWrong = this.shuffleArray(wrongIndices).slice(0, 2);

                const optBtns = document.querySelectorAll('.option-btn');
                shuffledWrong.forEach(idx => {
                    optBtns[idx].style.opacity = '0.25';
                    optBtns[idx].disabled = true;
                });
            }

            usePowerupTime() {
                if (!this.powerups.pTime) return;
                sound.playClick();
                this.powerups.pTime = false;
                this.renderPowerups();

                this.timer += 15;
                this.updateTimerDisplay();
            }

            usePowerupHint() {
                if (!this.powerups.pHint) return;
                sound.playClick();
                this.powerups.pHint = false;
                this.renderPowerups();

                const q = this.questions[this.currentIdx];
                const hintBox = document.getElementById('hint-box');
                document.getElementById('hint-text').innerText = q.hint;
                hintBox.classList.remove('hidden');
            }

            renderPowerups() {
                const b1 = document.getElementById('btn-powerup-5050');
                const b2 = document.getElementById('btn-powerup-time');
                const b3 = document.getElementById('btn-powerup-hint');

                b1.disabled = !this.powerups.p5050;
                b2.disabled = !this.powerups.pTime;
                b3.disabled = !this.powerups.pHint;
            }

            finishQuiz() {
                clearInterval(this.timerInterval);

                const totalQ = this.userAnswers.length;
                const correctCount = this.userAnswers.filter(a => a.isCorrect).length;
                const accuracy = totalQ > 0 ? Math.round((correctCount / totalQ) * 100) : 0;
                const totalTime = this.userAnswers.reduce((acc, curr) => acc + curr.timeTaken, 0);
                const avgTime = totalQ > 0 ? Math.round(totalTime / totalQ) : 0;

                const xpGained = (correctCount * 30) + (accuracy >= 80 ? 100 : 20);
                state.addXP(xpGained);

                state.stats.gamesPlayed++;
                state.stats.totalQuestions += totalQ;
                state.stats.correctAnswers += correctCount;

                this.userAnswers.forEach(ans => {
                    const cat = ans.question.category;
                    if (!state.stats.categoryScores[cat]) state.stats.categoryScores[cat] = 0;
                    if (ans.isCorrect) state.stats.categoryScores[cat]++;
                });

                state.checkBadges(accuracy, avgTime);

                const resultSummary = {
                    mode: this.mode === 'adventure' ? 'Mode Petualangan' : (this.mode === 'speedrun' ? 'Ujian Berwaktu' : 'Latihan Subjek'),
                    score: this.score,
                    correctCount: correctCount,
                    totalQ: totalQ,
                    accuracy: accuracy,
                    avgTime: avgTime,
                    xpGained: xpGained
                };

                app.renderResults(resultSummary);
                sheetService.sendResultToPraBab2(resultSummary);

                if (accuracy >= 70) {
                    app.triggerConfetti();
                }
            }

            shuffleArray(array) {
                const arr = [...array];
                for (let i = arr.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [arr[i], arr[j]] = [arr[j], arr[i]];
                }
                return arr;
            }
        }

            initKeyboardShortcuts() {
                document.addEventListener('keydown', (e) => {

                    const keyMap = { '1': 0, '2': 1, '3': 2, '4': 3, 'a': 0, 'b': 1, 'c': 2, 'd': 3 };
                    const key = e.key.toLowerCase();
                    if (keyMap.hasOwnProperty(key)) {
                        const btns = document.querySelectorAll('.option-btn');
                        if (btns[keyMap[key]] && !btns[keyMap[key]].disabled) {
                            game.handleAnswer(keyMap[key]);
                        }
                    }
                });
            }

            triggerConfetti() {
                const canvas = document.getElementById('confetti-canvas');
                if (!canvas) return;
                const ctx = canvas.getContext('2d');
                canvas.width = window.innerWidth;
                canvas.height = window.innerHeight;

                const particles = [];
                const colors = ['#3b82f6', '#10b981', '#f59e0b', '#ec4899', '#8b5cf6'];

                for (let i = 0; i < 80; i++) {
                    particles.push({
                        x: Math.random() * canvas.width,
                        y: Math.random() * canvas.height - canvas.height,
                        vx: (Math.random() - 0.5) * 3,
                        vy: Math.random() * 3 + 2,
                        color: colors[Math.floor(Math.random() * colors.length)],
                        size: Math.random() * 8 + 4
                    });
                }

                let frames = 0;
                function loop() {
                    ctx.clearRect(0, 0, canvas.width, canvas.height);
                    particles.forEach(p => {
                        p.x += p.vx;
                        p.y += p.vy;
                        ctx.fillStyle = p.color;
                        ctx.fillRect(p.x, p.y, p.size, p.size);
                    });
                    frames++;
                    if (frames < 120) {
                        requestAnimationFrame(loop);
                    } else {
                        ctx.clearRect(0, 0, canvas.width, canvas.height);
                    }
                }
                loop();
            }
